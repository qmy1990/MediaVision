cmake_minimum_required(VERSION 3.22)
project(MoonMediaVisionDemos LANGUAGES C CXX)
set(MEDIAVISION_SDK_ROOT "" CACHE PATH "Prebuilt SDK prefix containing lib/cmake/MediaVisionSDK")
if(NOT MEDIAVISION_SDK_ROOT)
  if(WIN32)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
      set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows/arm64/shared")
    elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
      set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows/x86/shared")
    else()
      set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows/x64/shared")
    endif()
  elseif(APPLE)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
      set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/macos/arm64/shared")
    else()
      set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/macos/x86_64/shared")
    endif()
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/linux/aarch64/shared")
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "armv7|armhf|^arm$")
    set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/linux/armv7/shared")
  elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
    set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/linux/x86/shared")
  else()
    set(MEDIAVISION_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/linux/x86_64/shared")
  endif()
endif()
find_package(MediaVisionSDK CONFIG REQUIRED
  PATHS "${MEDIAVISION_SDK_ROOT}/lib/cmake/MediaVisionSDK" NO_DEFAULT_PATH)
add_executable(mvs_demo_cli demos/cli/main.cpp)
target_link_libraries(mvs_demo_cli PRIVATE MediaVision::SDK)
target_compile_features(mvs_demo_cli PRIVATE cxx_std_17)
get_target_property(_mediavision_sdk_type MediaVision::SDK TYPE)
if(_mediavision_sdk_type STREQUAL "SHARED_LIBRARY")
  if(APPLE)
    set_target_properties(mvs_demo_cli PROPERTIES
      BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path")
  elseif(UNIX)
    set_target_properties(mvs_demo_cli PROPERTIES
      BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
  endif()
  add_custom_command(TARGET mvs_demo_cli POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      $<TARGET_FILE:MediaVision::SDK> $<TARGET_FILE_DIR:mvs_demo_cli>)
endif()
