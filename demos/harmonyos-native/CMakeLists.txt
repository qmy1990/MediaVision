cmake_minimum_required(VERSION 3.22)
project(MediaVisionHarmonyNative LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS" OR OHOS_PLATFORM_LEVEL LESS 20)
  message(FATAL_ERROR "HarmonyOS API20+ toolchain required")
endif()
get_filename_component(MVS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MVS_SDK_ROOT "${MVS_ROOT}/sdk/harmonyos" CACHE PATH "Binary HarmonyOS SDK directory")
set(MVS_LIBS "${MVS_SDK_ROOT}/${OHOS_ARCH}/shared/lib")
foreach(name mvs_sdk mvs_harmony)
  add_library(${name} SHARED IMPORTED)
  set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${MVS_LIBS}/lib${name}.so")
endforeach()
add_executable(mvs_harmony_native main.c)
target_compile_features(mvs_harmony_native PRIVATE c_std_11)
target_include_directories(mvs_harmony_native PRIVATE "${MVS_SDK_ROOT}/${OHOS_ARCH}/shared/include")
target_link_libraries(mvs_harmony_native PRIVATE mvs_harmony mvs_sdk)
target_link_options(mvs_harmony_native PRIVATE "-Wl,-rpath-link,${MVS_LIBS}")
