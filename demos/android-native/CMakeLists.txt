cmake_minimum_required(VERSION 3.22)
project(MediaVisionNativeConsumer LANGUAGES C)
if(NOT ANDROID)
  message(FATAL_ERROR "Use the Android NDK CMake toolchain")
endif()
get_filename_component(MVS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MVS_LIBS "${MVS_ROOT}/sdk/android/camera-native/${ANDROID_ABI}")
set(MVS_RUNTIME "${MVS_ROOT}/sdk/android/native-runtime/${ANDROID_ABI}")
foreach(name mvs_sdk mvs_inference)
  add_library(${name} SHARED IMPORTED)
  set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${MVS_LIBS}/lib${name}.so")
endforeach()
add_executable(mvs_native_inference main.c)
target_compile_features(mvs_native_inference PRIVATE c_std_11)
target_include_directories(mvs_native_inference PRIVATE "${MVS_ROOT}/include")
target_link_libraries(mvs_native_inference PRIVATE mvs_inference mvs_sdk m)
target_link_options(mvs_native_inference PRIVATE
  "-Wl,-rpath-link,${MVS_LIBS}" "-Wl,-rpath-link,${MVS_RUNTIME}"
  "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
